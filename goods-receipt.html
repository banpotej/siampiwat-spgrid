ตัวอย่าง DataGrid: รายการรับสินค้า (Goods Receipt)

คำอธิบายตัวอย่าง

ตัวอย่างตารางรายการรับสินค้า (Goods Receipt) จากข้อมูลสมมติ แสดงสถานะเอกสารทั้งแบบข้อความไทย และสถานะเอกสารภาษาอังกฤษด้วย badge ตัวเลข วันที่ และค่าว่าง (NULL) ของเอกสารฉบับร่างที่ยังไม่มีเลขที่

  • คลิกหัวคอลัมน์เพื่อเรียงลำดับ ค้นหาตามคอลัมน์ได้จากกล่องค้นหา
  • ดับเบิลคลิกแถว หรือกด Enter ที่แถวที่โฟกัส เพื่อดูรายละเอียด
  • ค่าที่เป็น NULL จะแสดงเป็น "-"
  • ด้านล่างสุดคือ Source Code ของตัวอย่างนี้ พร้อมปุ่มคัดลอก

การใช้งานในโค้ด

import { DataGrid, thaiLabels } from '/src/index.js';

const grid = new DataGrid(document.getElementById('grid'), {
  rowKey: 'id',
  rows, // เช่น { id: 240517, grNumber: 'GR26-031205', materialDoc: null, status: 'รออนุมัติ GR', ... }
  columns: [
    { key: 'grNumber', label: 'เลขที่ GR', sortable: true },
    { key: 'status', label: 'สถานะ', type: 'badge', badges: { ร่าง: { tone: 'neutral' } } },
    { key: 'orderNum', label: 'จำนวนรายการ', type: 'number' },
    { key: 'submittedDate', label: 'วันที่ส่งเอกสาร', type: 'date' },
  ],
  searchable: true,
  searchColumnSelect: true, // เลือกค้นเฉพาะคอลัมน์
  pageSize: 10,
  onRowActivate: ({ row }) => console.log(row), // ดับเบิลคลิก / Enter
  labels: thaiLabels,
});
  • ค่า null ใน rows (เช่น เลข GR ของร่าง) แสดงเป็น "–" เอง ไม่ต้องแปลงเป็นข้อความ
  • วันที่ส่งเป็น Date หรือ null ก็ได้ คอลัมน์ type: 'date' จัดรูปแบบให้
  • badges แมปค่าข้อความ (ทั้งไทยและอังกฤษ) เป็นสี tone ของ badge
  • ปรับความละเอียดของรายละเอียดแถวได้ด้วย onRowActivate (ตัวอย่างนี้เปิด modal)
