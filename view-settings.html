ตัวอย่าง: ตั้งค่าหน้าตาจากโค้ด


      

        

          คำอธิบายตัวอย่าง
          

            

              ค่าที่ผู้ใช้ปรับได้ใน modal ตั้งค่า (แถวสลับสี ตัดคำ เส้นตาราง ลักษณะเส้น ตรึงคอลัมน์
              และการจัดกลุ่ม) สั่งจากโค้ดได้ด้วย setViewSettings() และ
              setGroupBy() โดยไม่ต้องเปิด *Toggle
              ค่าที่ไม่ส่งมาจะคงเดิม และค่าที่ไม่ถูกต้องจะโยน error โดยไม่เปลี่ยนอะไรเลย ใส่
              *StorageKey เพื่อจำค่าไว้ใน localStorage (ลองรีโหลดหน้า)
            

            
// อ่านค่าปัจจุบัน
const { striped, wrapped, gridLines, lineStyle, freezeColumns } = grid.getViewSettings();

// ชุดค่าสำเร็จรูป: ส่งเฉพาะที่อยากเปลี่ยน
grid.setViewSettings({ striped: true, gridLines: 'both', lineStyle: 'dashed', freezeColumns: 2 });

// ค่าไม่ถูกต้อง: RangeError / TypeError และไม่มีอะไรเปลี่ยน
try {
  grid.setViewSettings({ striped: false, gridLines: 'diagonal' });
} catch (error) {
  console.warn(error.message);
}

// การจัดกลุ่มเป็นเมธอดของมันเอง
grid.setGroupBy(['department']);
