              สีเริ่มต้นของ SPGrid ไม่ต้องใส่ class ธีมใดๆ หัวตารางสีเทาเรียบ ขอบมน 8px
              และปุ่มสีทึบใช้คู่ class โทน + spg-action-solid ธีมอื่นเปิดเพิ่มได้ด้วย
              class เช่น spg-theme-lavender ที่ element ของตาราง
            

            
grid.classList.add('spg-theme-default');
new SPGrid(grid, { columns, rows, striped: true, gridLines: 'both' });

// ปุ่ม action สีทึบ: คู่ class โทน + spg-action-solid
{ name: 'edit', label: 'แก้ไข', className: 'spg-action-warning spg-action-solid' }
